    python3 -m venv .venv
. .venv/bin/activate
python3 -m pip install 'git+https://github.com/jonah-ux/atlas-agent-runtime.git@v0.2.0'
atlas demo --state ./atlas-demo-events.jsonl
atlas receipt demo-task --state ./atlas-demo-events.jsonl
